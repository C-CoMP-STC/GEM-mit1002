% Extra LaTeX preamble for the manifesto PDF build (see build.sh).
%
% LaTeX allows only 4 levels of nested itemize and 4 of enumerate by default,
% and errors with "Too deeply nested" beyond that. Outline-style markdown
% routinely goes deeper, so raise the limit and give each level its own marker.
%
% Requires the enumitem package:  sudo tlmgr install enumitem

\usepackage{enumitem}
\setlistdepth{9}

\setlist[itemize,1]{label=$\bullet$}
\setlist[itemize,2]{label=$\circ$}
\setlist[itemize,3]{label=$-$}
\setlist[itemize,4]{label=$\cdot$}
\setlist[itemize,5]{label=$\bullet$}
\setlist[itemize,6]{label=$\circ$}
\setlist[itemize,7]{label=$-$}
\setlist[itemize,8]{label=$\cdot$}
\setlist[itemize,9]{label=$\bullet$}
\renewlist{itemize}{itemize}{9}

\setlist[enumerate,1]{label=\arabic*.}
\setlist[enumerate,2]{label=\alph*.}
\setlist[enumerate,3]{label=\roman*.}
\setlist[enumerate,4]{label=\Alph*.}
\setlist[enumerate,5]{label=\arabic*.}
\setlist[enumerate,6]{label=\alph*.}
\setlist[enumerate,7]{label=\roman*.}
\setlist[enumerate,8]{label=\Alph*.}
\setlist[enumerate,9]{label=\arabic*.}
\renewlist{enumerate}{enumerate}{9}

% ---------------------------------------------------------------------------
% Wrap a long author list.
%
% LaTeX's default \maketitle sets the author block inside a tabular cell, and
% tabular cells cannot line-break -- so a nine-author list simply runs off the
% right edge of the page. This redefines \@maketitle to set the authors in a
% fixed-width centred parbox instead, which wraps normally. Spacing matches the
% article-class default, so nothing else about the title block changes.
%
% Adjust 0.85 if you want the author block narrower or wider.
\makeatletter
\renewcommand{\@maketitle}{%
  \null
  \vskip 2em%
  \begin{center}%
  \let\footnote\thanks
    {\LARGE \@title \par}%
    \vskip 1.5em%
    {\large \parbox{0.85\textwidth}{\centering \@author}\par}%
    \vskip 1em%
    {\large \@date}%
  \end{center}%
  \par
  \vskip 1.5em}
\makeatother
